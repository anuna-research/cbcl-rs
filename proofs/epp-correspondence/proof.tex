\section{Preliminaries (recap)}
\label{sec:prelim}
We recall, without proof, the CBCL results this development builds on; they are
established in the paper and the mechanized R1--R5 development.

\begin{definition}[Projection on protocols]
\label{def:proj}
$\mathrm{project}(P,r)$ maps a role-annotated protocol $P$ and role $r$ to a local
protocol whose steps are tagged from $r$'s viewpoint: a performative with
$\mathrm{from}=r$ is a \emph{Send}, one with $r\in\mathrm{to}$ a \emph{Recv}, and one in
which $r$ neither sends nor receives a \emph{bystander}, erased with its causal edges
spliced (transitively closed) through the erasure.
\end{definition}

\begin{definition}[Projectability]
\label{def:projectable}
$P$ is \emph{projectable onto $r$} iff $r$'s local protocol references, as a
predecessor, only performatives that $r$ itself sends or receives.
\end{definition}

\begin{definition}[Role-local verification]
\label{def:rolelocal}
On an incoming message $m$, the verifier checks \emph{role conformance}---a predicate on
immutable fields requiring $\mathrm{from}(m)$ and $\mathrm{to}(m)$ to match the
sender/recipient roles $P$ annotates for $\tau(m)$, returning $\mathit{Violation}$ on
mismatch---and runs the causal verifier against $\mathrm{project}(P,r)$ and $r$'s store.
The causal verifier returns $\mathit{Violation}$ for any \caused{} edge whose predecessor
type is not named in $\tau(m)$'s clause (\emph{no spurious predecessors}, as in the
reference implementation), so a $\mathit{Valid}$ message carries \emph{only} named, legal
predecessors.
Consequently a $P$-safe configuration is \emph{role-conformant}: every message's roles
match its performative's annotations.
\end{definition}

\noindent\emph{Remark (verdict timing).}
This development and its mechanization use \emph{resolved-first} semantics: no terminal
verdict is issued while any referenced predecessor is absent, so even a role-conformance
mismatch is reported as $\mathit{Violation}$ only once the message is resolved; before
that the verdict is $\mathit{Unknown}$.
The deployed verifier evaluates eagerly (valid-is-sticky) and may emit a provisional
$\mathit{Violation}$ later superseded by $\mathit{Valid}$; the two semantics coincide from
resolution onward, which is the regime all results below are stated in (closed
configurations and causally closed runs are fully resolved).
See ADR-602 in SPEC-014.

\begin{definition}[Valid-sticky verdict lattice]
\label{lat:lattice}
Verdicts are $\mathit{Unknown}$ (bottom) and the incomparable maximal $\mathit{Valid}$,
$\mathit{Violation}$ (a flat lattice; no top).
A message's verdict is fixed by its own \caused{} predecessors, which are immutable once
present (content addressing).
Under store growth a verdict moves up from $\mathit{Unknown}$ to a \emph{permanent}
$\mathit{Valid}$ or $\mathit{Violation}$; only $\mathit{Unknown}$ (a referenced
predecessor not yet present) is provisional.
\end{definition}

\begin{theorem}[Projectability $\equiv$ local verifiability]
\label{thm:equiv}
If $P$ is projectable onto $r$, then verification of any $r$-relevant message over
$\mathrm{project}(P,r)$ and $r$'s local store never returns $\mathit{Unknown}$ for a
predecessor $r$ cannot eventually observe.
\end{theorem}

\begin{proposition}[Monotonicity]
\label{prop:mono}
Role-local verification is monotone into the verdict lattice: $\mathit{Valid}$ and
$\mathit{Violation}$ are both stable under store growth; only $\mathit{Unknown}$ may
change (when a referenced predecessor arrives).
\end{proposition}

\begin{proposition}[DCFL preservation]
\label{prop:dcfl}
Each local protocol's accepted language is regular, hence DCFL; projection adds no
recogniser.
\end{proposition}

\noindent\emph{Remark (mechanization).}
Proposition~\ref{prop:dcfl} is mechanized at the safety level in the raw-edge regime
(\code{R6DCFLPreservation.lean}), with the deployed verifier's clause semantics
(disjunctive \code{Single}/\code{Any} pool, first-\code{(all \ldots)} fan-in,
undeclared performatives unconstrained). Two trace languages are proved regular, each
recognized exactly by an explicit finite automaton over the canonical seen-set
(sublists of the protocol's relevant names): the \emph{causal-order} arrival language
(\code{causalTrace\_regular}), and the order-free \emph{store} language
(\code{storeTrace\_regular}) — the language the valid-sticky verdicts
(Def.~\ref{lat:lattice}) actually accept, since out-of-order arrival is
$\mathit{Unknown}$ then $\mathit{Valid}$ and acceptance is ``every message eventually
$\mathit{Valid}$''. \code{causalTrace\_storeTrace} embeds the former in the latter.
Regular~$\subseteq$~DCFL is formal (\code{IsRegular.toTraceDCFL} wraps the automaton as
a stack-untouched deterministic pushdown machine). ``Projection adds no recogniser'' is
recorded as a definitional equality — the local protocol's recognizer \emph{is} the
same builder applied to the projected step list
(\code{projection\_adds\_no\_recogniser}); its content is that the protocol class is
closed under projection, and the substantive local statement is the regularity
instance at the projected protocol.

\begin{definition}[R6 well-formedness]
\label{def:r6}
A role-annotated protocol satisfies \emph{R6} iff it is role-complete, chooser-coherent,
projectable onto every role, \emph{causally local}---every endpoint role of a performative
$t$ is also an endpoint role of each predecessor type named in $t$'s \caused{} clause---%
every role is reachable from \code{begin}, and every choice is decided by a single agent.
(Causal locality is the multiparty addition: it makes every recorded dependency observable
to all of a message's endpoints, so the bystander-splice in Def~\ref{def:proj} is never
exercised between an endpoint performative and its predecessors.)
\end{definition}

\section{Formal Model}
\label{sec:model}

\subsection{Substrate}
A \emph{message} $m$ carries a content hash $h(m)$, a performative type $\tau(m)$, a
sender role $\mathrm{from}(m)$, a set of recipient roles $\mathrm{to}(m)$, and a set of
predecessor hashes $\mathrm{pred}(m)$; all references are thread-scoped.
A protocol $P$ (satisfying R6, Def~\ref{def:r6}) fixes, per performative, its predecessor
clauses (\code{(any \ldots)}/\code{(all \ldots)}) and sender/recipient roles.

\begin{definition}[Configuration]
\label{def:config}
A \emph{configuration} $C$ is a finite set of messages.
$C$ is \emph{closed} iff causally down-closed: for every $m\in C$ and
$\hbar\in\mathrm{pred}(m)$ there is $m'\in C$ with $h(m')=\hbar$.
\end{definition}

A configuration need not be closed: a dangling predecessor leaves the dependent message
at $\mathit{Unknown}$, never $\mathit{Violation}$.
Global executions of interest are closed configurations.

\subsection{Global validity}
We use two levels: a monotone safety core and a terminal completion predicate.

\begin{definition}[Safety]
\label{def:psafe}
A configuration $C$ is \emph{$P$-safe} iff no $m\in C$ has verdict $\mathit{Violation}$
under $P$ with store $C$; $\mathit{Unknown}$ is tolerated.
\end{definition}

\begin{proposition}[Safety is subset-closed]
\label{prop:safe-mono}
If $C\subseteq C'$ and $C'$ is $P$-safe then $C$ is $P$-safe.
\end{proposition}
\begin{proof}
A $\mathit{Violation}$ depends only on immutable fields of $m$ and on predecessors
present in the store (Def~\ref{def:rolelocal}); deleting messages can only turn a
resolved predecessor absent, i.e.\ $\mathit{Valid}/\mathit{Violation}\to\mathit{Unknown}$
(Def~\ref{lat:lattice}), never create a new $\mathit{Violation}$.
Hence no $m\in C$ is a $\mathit{Violation}$ under $C$.
\end{proof}

\begin{definition}[Completion]
\label{def:pcomplete}
A closed configuration $C$ is \emph{$P$-complete} iff it is $P$-safe and every obligation
is discharged: for each taken branch every required performative is present; every indexed
(\code{(all role[*])}) fan-in whose membership is sealed has all members present and
$\mathit{Valid}$; and every pooled (\code{(any role[*])}) obligation has at least one
occupant.
\end{definition}

\subsection{Local executions}
A message $m$ is \emph{$r$-relevant} iff $\mathrm{from}(m)=r$ or $r\in\mathrm{to}(m)$.

\begin{definition}[Run projection]
\label{def:run-proj}
$\mathrm{project}(C,r)$ is the set of $r$-relevant messages of $C$, each \emph{retaining
its raw} \caused{} (its $\mathrm{pred}$); no predecessor is rewritten.
$\mathrm{project}(P,r)$ is role $r$'s local protocol (Def~\ref{def:proj}).
\end{definition}

\begin{definition}[Local validity]
\label{def:localvalid}
A local run $L$ is \emph{locally $P$-safe for $r$} iff no $m\in L$ is $\mathit{Violation}$
under $\mathrm{project}(P,r)$ with store $L$ (Def~\ref{def:rolelocal});
\emph{locally complete for $r$} iff it additionally discharges every obligation of
$\mathrm{project}(P,r)$ restricted to $r$ (cf.\ Def~\ref{def:pcomplete}).
\end{definition}

By Theorem~\ref{thm:equiv}, when $P$ satisfies R6 a locally $P$-safe run carries no
spurious $\mathit{Unknown}$ for a predecessor $r$ cannot observe.

\subsection{Gluing}

\begin{definition}[Family and compatibility]
\label{def:compatible}
A \emph{family} $F=\{L_r\}_{r}$ has one local run per role, each $L_r$ a set of
$r$-relevant messages.
$F$ is \emph{compatible} iff:
\emph{(Agreement)} messages occurring in two runs with equal hash are identical; and
\emph{(Coverage)} for every $m\in L_r$:
\textup{(a)} $m\in L_{r'}$ for every endpoint $r'\in\{\mathrm{from}(m)\}\cup\mathrm{to}(m)$;
and
\textup{(b)} every \caused{} predecessor of $m$ is itself in $L_r$---each local run is
\emph{causally closed};
and \emph{(Seal)} all runs agree on the \emph{cast}---the assignment of agents to roles
fixed at the thread's causal root, including the sealed membership of each indexed role.
\end{definition}

\noindent
Coverage(b) is exactly the closure that $\mathrm{project}(C,r)$ enjoys by Local
resolvability (Lemma~\ref{lem:localres}); requiring it asks no more than that each run be a
genuine projection.

\begin{definition}[Gluing]
\label{def:glue}
$\mathrm{glue}(F)=\bigcup_r L_r$, deduplicated by hash.
\end{definition}

\begin{lemma}[Glue is well-defined and closed]
\label{lem:glue-closed}
For a compatible family $F$, $\mathrm{glue}(F)$ is a configuration and is closed.
\end{lemma}
\begin{proof}
Well-defined: equal-hash messages are identical by Agreement, so the deduplicated union
is a set of messages.
Closed: for $m\in\mathrm{glue}(F)$, $m\in L_r$ for some $r$; by Coverage(b) every
\caused{} predecessor of $m$ lies in $L_r\subseteq\mathrm{glue}(F)$.
\end{proof}

\begin{lemma}[Local resolvability]
\label{lem:localres}
Let $C$ be a $P$-safe \emph{closed} configuration of a protocol satisfying R6, and $m\in C$.
For every endpoint $r$ of $m$ (i.e.\ $\mathrm{from}(m)=r$ or $r\in\mathrm{to}(m)$), every
\caused{} predecessor of $m$ is $r$-relevant, hence lies in $\mathrm{project}(C,r)$.
\end{lemma}
\begin{proof}
Let $b$ be a predecessor of $m$ ($h(b)\in\mathrm{pred}(m)$).
Since $C$ is closed, every predecessor of $m$ is present, so $m$'s verdict is decided;
$P$-safety excludes $\mathit{Violation}$, hence $m$ is $\mathit{Valid}$.
By Def~\ref{def:rolelocal} a $\mathit{Valid}$ message has \emph{only} named, legal
predecessors, so every $\tau(b)$ in $m$'s \caused{} is a predecessor type of $\tau(m)$
named in its clause.
By causal locality (R6, Def~\ref{def:r6}), since $r$ is an endpoint of $\tau(m)$, $r$ is an
endpoint role of $\tau(b)$.
Role-conformance of the $P$-safe $C$ (Def~\ref{def:rolelocal}) makes $\mathrm{from}(b)$ and
$\mathrm{to}(b)$ match $\tau(b)$'s annotations, so $r$ is an endpoint of $b$; thus $b$ is
$r$-relevant and $b\in\mathrm{project}(C,r)$.
\end{proof}

\begin{lemma}[Reconciliation: verdicts agree]
\label{lem:reconcile}
Let $P$ satisfy R6.
\textup{(i)} For a $P$-safe closed configuration $C$, role $r$, and
$m\in\mathrm{project}(C,r)$, role-local verification of $m$ over $\mathrm{project}(P,r)$ with
store $\mathrm{project}(C,r)$ returns the same verdict as verification of $m$ over $P$ with
store $C$.
\textup{(ii)} For a compatible family $F$ with each $L_r$ locally $P$-safe, role $r$, and
$m\in L_r$, verification of $m$ over store $\mathrm{glue}(F)$ returns the same verdict as
over store $L_r$.
\end{lemma}
\begin{proof}
Verification of $m$ reads only $m$'s immutable fields and the messages named by its
\caused{} (Def~\ref{def:rolelocal}, Def~\ref{lat:lattice}); role conformance and the
predecessor-type checks, composed by meet/join for \code{(all \ldots)}/\code{(any \ldots)},
depend only on those.
\textup{(i)} By Local resolvability (Lemma~\ref{lem:localres}) every \caused{} predecessor
of $m$ lies in $\mathrm{project}(C,r)\subseteq C$ with the same hash, so both verifications
read identical messages and return the same verdict.
\textup{(ii)} By Coverage(b) every \caused{} predecessor of $m$ lies in
$L_r\subseteq\mathrm{glue}(F)$, and by Agreement the shared messages are identical, so
neither store offers $m$ a predecessor the other lacks; the verdicts coincide.
\end{proof}

\begin{lemma}[Completion reconciliation]
\label{lem:complete-reconcile}
Let $P$ satisfy R6.
\textup{(i)} If $C$ is a $P$-complete closed configuration then each $\mathrm{project}(C,r)$
is locally complete.
\textup{(ii)} If $F$ is a compatible family with each $L_r$ locally complete then
$\mathrm{glue}(F)$ is $P$-complete.
\end{lemma}
\begin{proof}
Every obligation of $\mathrm{project}(P,r)$ is the $r$-restriction of an obligation of $P$,
and conversely each obligation of $P$ is, for some endpoint role $r$, an obligation of
$\mathrm{project}(P,r)$; sealed memberships agree across $C$ / the family (Seal).
\textup{(i)} Safety holds by Lemma~\ref{lem:reconcile}(i).
A required performative $t$ with endpoint $r$ is present in $C$ (completeness) and
$r$-relevant, so it lies in $\mathrm{project}(C,r)$ (Lemma~\ref{lem:localres}); a sealed
\code{(all role[*])} fan-in $r$ participates in has all its members present in $C$ and,
each being $r$-relevant, in $\mathrm{project}(C,r)$; a pooled \code{(any role[*])} obligation
keeps its occupant.
So every obligation of $\mathrm{project}(P,r)$ is discharged.
\textup{(ii)} $\mathrm{glue}(F)$ is $P$-safe (Lemma~\ref{lem:reconcile}(ii)) and closed
(Lemma~\ref{lem:glue-closed}).
For any obligation of $P$, pick an endpoint role $r$ for which it is an obligation of
$\mathrm{project}(P,r)$; local completeness of $L_r$ discharges it there, and the witnessing
messages lie in $L_r\subseteq\mathrm{glue}(F)$, with sealed memberships fixed by (Seal).
Hence $\mathrm{glue}(F)$ is $P$-complete.
\end{proof}

\section{The correspondence}
\label{sec:corr}

\begin{theorem}[EPP correspondence]
\label{thm:epp}
Let $P$ satisfy R6.
\begin{enumerate}
\item[(1)] \emph{Soundness.} If $C$ is a $P$-safe (resp.\ $P$-complete) closed
configuration, then for every role $r$, $\mathrm{project}(C,r)$ is locally $P$-safe (resp.\
locally complete) for $r$.
\item[(2)] \emph{Completeness.} If $F=\{L_r\}$ is a \emph{compatible} family with each
$L_r$ locally $P$-safe (resp.\ locally complete), then $\mathrm{glue}(F)$ is a $P$-safe
(resp.\ $P$-complete) closed configuration.
\item[(3)] \emph{Exactness.} $\mathrm{project}$ and $\mathrm{glue}$ are mutually inverse on
these objects: $\mathrm{glue}(\{\mathrm{project}(C,r)\}_r)=C$ for $P$-safe closed $C$, and
$\mathrm{project}(\mathrm{glue}(F),r)=L_r$ for compatible $F$.
Hence $P$-safe (resp.\ $P$-complete) closed configurations and compatible families of
locally $P$-safe (resp.\ locally complete) runs are in bijection---the endpoints realise
\emph{exactly} $P$.
\end{enumerate}
\end{theorem}

\begin{proof}[Proof of (1)]
Let $r$ be a role and $m\in\mathrm{project}(C,r)$.
By Lemma~\ref{lem:reconcile}(i) the verdict of $m$ over $\mathrm{project}(P,r)$ with store
$\mathrm{project}(C,r)$ equals its verdict over $P$ with store $C$, which is
non-$\mathit{Violation}$ since $C$ is $P$-safe; so $\mathrm{project}(C,r)$ is locally
$P$-safe.
For the $P$-complete case, local completeness is Lemma~\ref{lem:complete-reconcile}(i).
\end{proof}

\begin{proof}[Proof of (2)]
$\mathrm{glue}(F)$ is closed by Lemma~\ref{lem:glue-closed}.
For $m\in\mathrm{glue}(F)$, $m\in L_r$ for some $r$; by Lemma~\ref{lem:reconcile}(ii) the
verdict of $m$ over $\mathrm{glue}(F)$ equals its verdict over $L_r$, which is
non-$\mathit{Violation}$ since $L_r$ is locally $P$-safe; so $\mathrm{glue}(F)$ is $P$-safe.
For the $P$-complete case, Lemma~\ref{lem:complete-reconcile}(ii).
\end{proof}

For exactness we record two facts about projections, both immediate from the definitions.

\begin{lemma}[Projection family is compatible]
\label{lem:proj-compat}
For a $P$-safe closed configuration $C$, the family $\{\mathrm{project}(C,r)\}_r$ is
compatible.
\end{lemma}
\begin{proof}
Each $\mathrm{project}(C,r)$ selects $r$-relevant messages of $C$ retaining their hashes, so
equal-hash messages across projections are the same message of $C$ (Agreement).
Coverage(a): if $m\in\mathrm{project}(C,r)$ and $r'$ is an endpoint of $m$ then $m$ is
$r'$-relevant and $m\in C$, so $m\in\mathrm{project}(C,r')$.
Coverage(b): Lemma~\ref{lem:localres}.
(Seal): all projections inherit $C$'s single cast.
\end{proof}

\begin{lemma}[Round-trip]
\label{lem:roundtrip}
For a closed configuration $C$, $\mathrm{glue}(\{\mathrm{project}(C,r)\}_r)=C$; and for a
compatible family $F$, $\mathrm{project}(\mathrm{glue}(F),r)=L_r$ for every role $r$.
\end{lemma}
\begin{proof}
$\supseteq$/$=$ for the first: every $m\in C$ has the endpoint $\mathrm{from}(m)$, so
$m\in\mathrm{project}(C,\mathrm{from}(m))\subseteq\mathrm{glue}$; conversely each projection
is a subset of $C$, and dedup by hash returns exactly $C$.
For the second: $\mathrm{project}(\mathrm{glue}(F),r)$ is the set of $r$-relevant messages
of $\mathrm{glue}(F)$.
Any such $m$ lies in some $L_{r'}$ and, being $r$-relevant, in $L_r$ by Coverage(a);
conversely $L_r\subseteq\mathrm{glue}(F)$ and its messages are $r$-relevant, so they are
retained by $\mathrm{project}(\cdot,r)$.
\end{proof}

\begin{proof}[Proof of (3)]
$\mathrm{glue}(\{\mathrm{project}(C,r)\}_r)=C$ and
$\mathrm{project}(\mathrm{glue}(F),r)=L_r$ are Lemma~\ref{lem:roundtrip};
Lemma~\ref{lem:proj-compat} makes the forward image a compatible family (a $P$-complete
configuration is $P$-safe and closed, so the lemma applies to that class too), and parts
(1)/(2) carry validity in both directions, so the two maps are mutually inverse bijections
between the stated classes.
\end{proof}

\section{Notes}
\label{sec:notes}
This development is standalone; ported into the paper, the recap (\S\ref{sec:prelim}) maps
onto the existing labelled results and Theorem~\ref{thm:epp} replaces the former conjecture.
\emph{Causal locality} (R6, Def~\ref{def:r6}) is the multiparty contribution: it is what
makes every recorded dependency observable to all of a message's endpoints, and the
implementation already enforces its type-level shadow (projectability, no spurious
predecessors).
Multi-recipient messages ($\mathrm{to}(m)$ a set) and multi-occupant roles (pooled/indexed)
are handled throughout: the verdict-based reconciliation checks the actual
\code{(any)}/\code{(all)} edges identically locally and globally, so no set-equality
obligation arises.
The completeness direction is conditioned on \emph{compatibility} (Def~\ref{def:compatible}),
a meta-level gluing precondition; making it endpoint-checkable is future work.
Theorem~\ref{thm:epp} is mechanized in Lean~4 (no \code{sorry}; axioms \code{propext},
\code{Classical.choice}, \code{Quot.sound}, enforced in CI by the project axiom audit) at
the abstract-model level, on top of the R1--R5 development; instantiating the clause
language and the concrete message types remains.
Two mechanization notes.
First, the mechanization projects the \emph{store} and verifies both sides against the
global $P$; the protocol side of Def~\ref{def:proj} enters through an agreement
interface (\code{AgreesOnRole}, \code{ProtocolProjection.lean}): any projected protocol
that agrees with $P$ on $r$-endpoint performatives assigns $r$-relevant messages
identical verdicts (\code{local\_protocol\_verification\_agrees}), and causal locality
(R6, Def~\ref{def:r6}) is exactly what makes the bystander-erasure/splice of
Def~\ref{def:proj} leave $r$-endpoint performatives untouched and hence satisfy the
interface; mechanizing the splice itself would require a concrete clause syntax, which
the abstract model deliberately omits.
Second, obligations (Def~\ref{def:pcomplete}) are mechanized as witness-\emph{sets}
discharged by any single present-and-$\mathit{Valid}$ member, so pooled
\code{(any role[*])} obligations are genuinely disjunctive (different local runs may
discharge one via different occupants); requiring $\mathit{Valid}$ rather than bare
presence for plain required performatives is vacuous on the correspondence's domain,
where presence implies validity (\code{valid\_of\_safe\_closed}).
